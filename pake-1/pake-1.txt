  (defrole init
    (vars (a b name) (pw data) (alpha beta rndx))
    (trace (send (exp (gen) alpha)) (recv (exp (gen) beta))
      (send
        (enc "hello" b "I am" a
          (hash pw a b (exp (gen) alpha) (exp (gen) beta)
            (exp (gen) (mul alpha beta)))))
      (recv
        (enc "got it" a "I am" b
          (hash pw a b (exp (gen) alpha) (exp (gen) beta)
            (exp (gen) (mul alpha beta)))))))

          (hash pw a b (exp (gen) alpha) (exp (gen) beta)
            (exp (gen) (mul alpha beta)))))))
